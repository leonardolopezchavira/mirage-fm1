# power-on: the boot ROM tunes the filters and loads OS 3.2 and the first sounds from the built-in disk
wait 9000
expect booted 1
expect motor 0
shot boot.png
# a key plays: F3 + 7 semitones = C4
wav note.wav
key 7 down 100
wait 600
key 7 up
wait 300
wavstop
expect peak_db_min -30
# a chord over MIDI, as from a keyboard
midi 90 30 64

wait 500

wait 300
# K1 picks parameter 38 (filter cutoff); when it rests, PARAM 3 8 VALUE goes to the Mirage, then K2 steps it down

wait 1500

turn K2 -2
wait 800

expect seg1 191
tap HOME

expect seg1 127
